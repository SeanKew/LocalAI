cmake_minimum_required(VERSION 3.22.1)
project(localai_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(MNN_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}")
set(MNN_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include")
foreach(required_file "${MNN_LIB_DIR}/libMNN.so" "${MNN_LIB_DIR}/libllm.so" "${MNN_INCLUDE_DIR}/llm/llm.hpp")
    if(NOT EXISTS "${required_file}")
        message(FATAL_ERROR "Missing MNN dependency: ${required_file}. See README.md.")
    endif()
endforeach()

add_library(MNN SHARED IMPORTED)
set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${MNN_LIB_DIR}/libMNN.so")
add_library(llm SHARED IMPORTED)
set_target_properties(llm PROPERTIES IMPORTED_LOCATION "${MNN_LIB_DIR}/libllm.so")
add_library(mnnllm_jni SHARED llm_jni.cpp)
target_include_directories(mnnllm_jni PRIVATE "${MNN_INCLUDE_DIR}")
target_link_libraries(mnnllm_jni PRIVATE llm MNN log android)
